AFRAME.registerComponent('physics-grabber', { schema: { throwMultiplier: { type: 'number', default: 1.4 } }, init: function () { this.grabbedEl = null; this.targetEl = null; this.positions = []; this.handWorldPos = new THREE.Vector3(); this.handWorldQuat = new THREE.Quaternion(); // Vecteurs pour préserver le décalage relatif this.holdOffsetPos = new THREE.Vector3(); this.holdOffsetQuat = new THREE.Quaternion(); this.targetWorldPos = new THREE.Vector3(); this.targetWorldQuat = new THREE.Quaternion(); // Détection laser (1.5 m) this.el.addEventListener('raycaster-intersection', (e) => { const valid = e.detail.els.filter(el => el.classList.contains('grabbable')); if (valid.length > 0) this.targetEl = valid[0]; }); this.el.addEventListener('raycaster-intersection-cleared', () => { this.targetEl = null; }); // Début de la saisie : calcul de l'écart initial (offset) const onGrab = () => { if (!this.targetEl || this.grabbedEl) return; this.grabbedEl = this.targetEl; this.positions = []; // Positions et rotations mondiales actuelles this.el.object3D.getWorldPosition(this.handWorldPos); this.el.object3D.getWorldQuaternion(this.handWorldQuat); this.grabbedEl.object3D.getWorldPosition(this.targetWorldPos); this.grabbedEl.object3D.getWorldQuaternion(this.targetWorldQuat); // Calcul du décalage relatif exprimé dans le repère local de la main const invHandQuat = this.handWorldQuat.clone().invert(); this.holdOffsetPos.subVectors(this.targetWorldPos, this.handWorldPos).applyQuaternion(invHandQuat); this.holdOffsetQuat.copy(invHandQuat).multiply(this.targetWorldQuat); }; // Relâchement et lancer const onRelease = () => { if (!this.grabbedEl) return; const body = this.grabbedEl.body; if (body && this.positions.length >= 2) { const oldest = this.positions[0]; const newest = this.positions[this.positions.length - 1]; const totalDt = (newest.time - oldest.time) / 1000; if (totalDt > 0.01) { const vx = ((newest.pos.x - oldest.pos.x) / totalDt) * this.data.throwMultiplier; const vy = ((newest.pos.y - oldest.pos.y) / totalDt) * this.data.throwMultiplier; const vz = ((newest.pos.z - oldest.pos.z) / totalDt) * this.data.throwMultiplier; body.velocity.set(vx, vy, vz); } } this.grabbedEl = null; this.positions = []; }; this.el.addEventListener('triggerdown', onGrab); this.el.addEventListener('gripdown', onGrab); this.el.addEventListener('triggerup', onRelease); this.el.addEventListener('gripup', onRelease); }, tick: function (time, delta) { this.el.object3D.getWorldPosition(this.handWorldPos); this.el.object3D.getWorldQuaternion(this.handWorldQuat); // Historique glissant pour le lancer this.positions.push({ pos: this.handWorldPos.clone(), time: performance.now() }); if (this.positions.length > 6) { this.positions.shift(); } // Déplacement de l'objet avec conservation de l'offset if (this.grabbedEl && this.grabbedEl.body) { const body = this.grabbedEl.body; // Applique la rotation de la main à l'offset initial const currentOffset = this.holdOffsetPos.clone().applyQuaternion(this.handWorldQuat); const newPos = this.handWorldPos.clone().add(currentOffset); // Met à jour la position du corps Cannon.js body.position.set(newPos.x, newPos.y, newPos.z); // Met à jour la rotation en combinant celle de la main et l'orientation initiale const newQuat = this.handWorldQuat.clone().multiply(this.holdOffsetQuat); body.quaternion.set(newQuat.x, newQuat.y, newQuat.z, newQuat.w); body.velocity.set(0, 0, 0); body.angularVelocity.set(0, 0, 0); } } });
